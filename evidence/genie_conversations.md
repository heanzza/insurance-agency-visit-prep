```sql
SELECT
  `p`.`agency_name`,
  SUM(`k`.`pwaiver_attached_policies`) AS `attached_policies`,
  SUM(`k`.`pwaiver_eligible_policies`) AS `eligible_policies`,
  ROUND(100 * TRY_DIVIDE(SUM(`k`.`pwaiver_attached_policies`), SUM(`k`.`pwaiver_eligible_policies`)), 1) AS `pwaiver_rate_pct`
FROM `jp_fsi_catalog`.`agency_visit_prep`.`gold_agency_monthly_kpi` AS `k`
INNER JOIN `jp_fsi_catalog`.`agency_visit_prep`.`gold_visit_priority` AS `p`
  ON `k`.`agency_id` = `p`.`agency_id`
WHERE `k`.`fiscal_year` = 2026
  AND `p`.`agency_name` IS NOT NULL
GROUP BY `p`.`agency_name`
HAVING SUM(`k`.`pwaiver_eligible_policies`) >= 20
ORDER BY `pwaiver_rate_pct` ASC
LIMIT 10
```
